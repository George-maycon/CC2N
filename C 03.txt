float distanciaPQ(float pX, float pY,
                   float qX, float qY);


 float distanciaPQ(float pX, float pY,
                   float qX, float qY){
          float distancia;
          distancia  = sqrt(pow(pX - qX,2) + pow(pY - qY,2));
          return distancia;

}
